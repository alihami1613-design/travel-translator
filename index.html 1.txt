  function speak(text, lang = voiceLang.value, rate = currentRate) {
    if (!("speechSynthesis" in window)) {
      statusMsg.textContent = "مرورگر از صدا پشتیبانی نمی‌کند.";
      return;
    }
    window.speechSynthesis.cancel();
    const utter = new SpeechSynthesisUtterance(text);
    utter.lang = lang;
    utter.rate = rate;
    const voices = window.speechSynthesis.getVoices();
    const matched = voices.find((v) => v.lang && v.lang.toLowerCase().includes(lang.toLowerCase().slice(0, 2)));
    if (matched) utter.voice = matched;
    utter.onstart = () => { statusMsg.textContent = `در حال پخش صدا (${lang})...`; };
    utter.onend = () => { statusMsg.textContent = "پخش صدا پایان یافت."; };
    window.speechSynthesis.speak(utter);
  }

  async function fetchOnlineTranslation(text, dest) {
    try {
      const url = `https://api.mymemory.translated.net/get?q=${encodeURIComponent(text)}&langpair=autodetect|${dest}`;
      const res = await fetch(url);
      if (!res.ok) return null;
      const data = await res.json();
      const translated = data && data.responseData && data.responseData.translatedText;
      if (translated && !translated.toLowerCase().includes("mymemory warning")) {
        return translated;
      }
      return null;
    } catch (_) {
      return null;
    }
  }

  function renderWords(text) {
    wordsGrid.innerHTML = "";
    const words = text.trim().split(/\s+/).filter(Boolean);
    words.forEach((w) => {

      pill.type = "button";
      pill.className = "word-pill";
      pill.textContent = w;
      pill.addEventListener("click", () => speak(w, voiceLang.value, currentRate));
      wordsGrid.appendChild(pill);
    });
  }

  async function translateText() {
    const text = inputText.value.trim();
    if (!text) {
      statusMsg.textContent = "لطفاً عبارتی وارد کنید.";
      return;
    }
    const dest = targetLang.value;
    phoneticOutput.textContent = "متن اصلی: " + text;
    renderWords(text);

    statusMsg.textContent = "در حال ترجمه آنلاین...";
    const online = await fetchOnlineTranslation(text, dest);
    if (online) {
      translatedOutput.textContent = online;
      statusMsg.textContent = "ترجمه انجام شد.";
    } else {
      translatedOutput.textContent = "خطا در اتصال به سرویس ترجمه آنلاین.";
      statusMsg.textContent = "لطفاً اتصال اینترنت را بررسی کنید.";
    }
  }

      recognition.lang = voiceLang.value;
      recognition.start();
    }
  });

  const STORAGE_KEY = "travel_speaker_saved_v2";
  function getSaved() {
    try {
      return JSON.parse(localStorage.getItem(STORAGE_KEY) || "[]");
    } catch (_) {
      return [];
    }
  }

  function setSaved(items) {
    localStorage.setItem(STORAGE_KEY, JSON.stringify(items));
    document.getElementById("savedCount").textContent = items.length;
  }

  function renderSaved() {
    const list = document.getElementById("savedList");
    const items = getSaved();
    list.innerHTML = "";
    if (!items.length) {
      list.innerHTML = '<p style="font-size:13px;color:var(--muted);">هنوز عبارتی ذخیره نشده است.</p>';
      return;
    }
    items.forEach((item, idx) => {
      const row = document.createElement("div");
      row.className = "saved-item";
      row.innerHTML = `
        <div>
          <div class="orig">${item.orig}</div>
          <div class="trans">${item.trans}</div>
        </div>
        <div style="display:flex;gap:6px;">
          <button class="chip" type="button" data-play="${idx}">پخش</button>
          <button class="chip" type="button" data-del="${idx}">حذف</button>
        </div>
      `;
      row.querySelector("[data-play]").addEventListener("click", () => speak(item.trans, voiceLang.value, currentRate));
      row.querySelector("[data-del]").addEventListener("click", () => {
        const next = getSaved().filter((_, i) => i !== idx);
        setSaved(next);
        renderSaved();
      });
      list.appendChild(row);
    });
  }

  document.getElementById("saveBtn").addEventListener("click", () => {
    const orig = inputText.value.trim();
    const trans = translatedOutput.textContent.trim();
    if (!orig || !trans) return;
    const items = getSaved();
    items.unshift({ orig, trans });
    setSaved(items.slice(0, 100));
    statusMsg.textContent = "عبارت ذخیره شد.";
  });

  setSaved(getSaved());
  updateSamples();
  translateText();
